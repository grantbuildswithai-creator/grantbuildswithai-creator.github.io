];

export function sectionColor(id: SectionId) {
  return SECTIONS.find((s) => s.id === id)!.color;
}

export type VisibleBuild = Build & { isNew: boolean; ready: boolean };

function isReady(b: Build, status: Build['status']) {
  if (status === 'soon') return false;
  if (b.kind === 'link') return true;
  return !!b.Screen && (!b.needsServer || hasServer);
}

// Everything after applying the remote switches, and after marking server-only builds
// "Soon" when no server is set up (e.g. the web version before the worker is deployed).
export function useBuilds(): VisibleBuild[] {
  const config = useRemoteConfig();
  return BUILDS.filter(
    (b) =>
      !config.builds?.[b.id]?.hidden &&
      (!b.platforms || b.platforms.includes(Platform.OS as 'android' | 'ios' | 'web')),
  ).map((b) => {
    const remote = config.builds?.[b.id];
    const status = remote?.status ?? b.status;
    const ready = isReady(b, status);
    return { ...b, status: ready ? status : 'soon', isNew: !!remote?.isNew, ready };
  });
}

// In-app builds only: links don't have pages of their own.
export function useAppBuild(id: string | undefined) {
  const build = useBuilds().find((b) => b.id === id);
  return build?.kind === 'app' ? build : undefined;
}
